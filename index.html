function lockAndGenerateBatch() { try { const unapproved = stagedMatches.filter(m => !m.approved); if (unapproved.length > 0) { if (!confirm(`You have ${unapproved.length} unapproved match(es). Lock and generate batch anyway?`)) return; } let tsvRows = []; stagedMatches.forEach((m, idx) => { const timestamp = new Date().toISOString(); const headerTitle = m.leagueName || (fetchedSheetMatches.length > 0 && fetchedSheetMatches[0] && fetchedSheetMatches[0].leagueName) || "8-BALL CASH LEAGUE"; const homeName = m.homeTeam || m.home || "Unknown Home"; const visitName = m.visitTeam || m.visit || m.visitorTeam || "Unknown Visit"; let hScore = "0", vScore = "0"; if (m.scoreText && typeof m.scoreText === 'string' && m.scoreText.includes('-')) { const parts = m.scoreText.split('-'); hScore = parts[0].trim(); vScore = parts[1].trim(); } const games = Array.isArray(m.individualGames) && m.individualGames.length > 0 ? m.individualGames : [{ gameNum: "Overall", homePlayer: homeName, visitPlayer: visitName, homeNP: false, homeWin: false, homeTR: false, home8BB: false, visitNP: false, visitWin: false, visitTR: false, visit8BB: false }]; games.forEach(game => { tsvRows.push([ timestamp, headerTitle, homeName, hScore, visitName, vScore, game.gameNum || "Gm", game.homePlayer || "", game.homeNP ? "Yes" : "No", game.homeWin ? "Win" : "", game.homeTR ? "TR" : "", game.home8BB ? "8BB" : "", game.visitPlayer || "", game.visitNP ? "Yes" : "No", game.visitWin ? "Win" : "", game.visitTR ? "TR" : "", game.visit8BB ? "8BB" : "", m.homeSig || "", m.visitSig || "" ].join('\t')); }); }); finalizedTSV = tsvRows.join('\n'); const outputBox = document.getElementById('tsv-output-box'); if (outputBox) outputBox.innerText = finalizedTSV; const sendBtn = document.getElementById('send-sheet-btn'); if (sendBtn) { sendBtn.disabled = false; sendBtn.style.opacity = '1'; } switchTabByElement(document.getElementById('btn-tab-sheets'), 'tab-sheets'); } catch (err) { alert('Error generating batch: ' + err.message); console.error("Lock batch error details:", err); } } function copyTSVToClipboard() { if (!finalizedTSV) { alert('No TSV payload available yet.'); return; } navigator.clipboard.writeText(finalizedTSV).then(() => { alert('TSV batch copied to clipboard successfully!'); }).catch(err => { const area = document.createElement('textarea'); area.value = finalizedTSV; document.body.appendChild(area); area.select(); document.execCommand('copy'); document.body.removeChild(area); alert('TSV batch copied to clipboard!'); }); } async function sendDataToSheet() { if (!resolvedMainUrl) { alert('Please connect to a league first to load the Main Sheet target.'); return; } if (!finalizedTSV) { alert('No locked TSV batch available to send.'); return; } document.getElementById('send-sheet-btn').innerText = 'SENDING...'; try { await fetch(resolvedMainUrl, { method: 'POST', mode: 'no-cors', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: finalizedTSV }); alert('Batch successfully sent to Main Sheet (Approved Matches)!'); document.getElementById('send-sheet-btn').innerText = 'Sent Successfully!'; } catch (err) { alert('Error sending batch to sheet: ' + err.message); document.getElementById('send-sheet-btn').innerText = 'Send Batch to Main Sheet'; } }
